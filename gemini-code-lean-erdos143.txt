import Mathlib.Analysis.Asymptotics.Asymptotics
import Mathlib.MeasureTheory.Measure.Lebesgue.Basic

open Set MeasureTheory Real

/-- 
  Erdős Problem 143: 
  A set A ⊂ (1, ∞) s.t. ∀ x ≠ y ∈ A and k ≥ 1, |kx - y| ≥ 1.
-/
def ErdosCondition (A : Set ℝ) : Prop :=
  (∀ x ∈ A, x > 1) ∧ 
  (∀ x ∈ A, ∀ y ∈ A, ∀ k : ℕ, k ≥ 1 → x ≠ y → |(k : ℝ) * x - y| ≥ 1)

/-- The logarithmic sum of the set A up to n -/
noncomputable def logSum (A : Set ℝ) (n : ℝ) : ℝ :=
  ∑' x, if x ∈ A ∧ x < n then 1/x else 0

  Theorem: ErdosCondition implies sparsity (o(log n))
